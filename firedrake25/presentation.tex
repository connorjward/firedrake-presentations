\begin{frame}
	\frametitle{How did we get here?}

	\begin{itemize}
		\item Firedrake needs a particular environment of compiled dependencies
		\item Most users are unfamiliar with compiling complex software
		\item HPC makes this even harder
		\item Hence \texttt{firedrake-install}
	\end{itemize}

	\pause

	But.

	\pause

	\begin{itemize}
		\item \texttt{firedrake-install} was not a robust solution
		\item Install failures were extremely hard to debug
	\end{itemize}
\end{frame}

\begin{frame}
	\frametitle{What happened?}

	\begin{itemize}
		\item \texttt{pip install libsupermesh}
		\item \texttt{pip install firedrake}
		\item 3 stage install with \texttt{firedrake-configure} (intentionally stupid)
		% this required a CI rewrite and threw up a lot of bugs
		\item OpenMPI instead of MPICH (\texttt{firedrake-run-split-tests})
		\item \texttt{firedrake-check}
		\item \texttt{--netgen} etc into \texttt{pyproject.toml}
		\item New Docker images
	\end{itemize}
\end{frame}

\begin{frame}
	\frametitle{Success stories}

	\begin{itemize}
		\item Much easier HPC install
		\item Other packaging systems (PETSc, NixOS)
	\end{itemize}
\end{frame}

  # Only test Gusto in real mode
  if: |
    inputs.run_tests &&

    matrix.arch == 'default'
  run: |
    . venv/bin/activate
    git clone --depth 1 https://github.com/firedrakeproject/gusto.git gusto-repo
    pip install --verbose ./gusto-repo
    python -m pytest -n 8 --verbose \
      gusto-repo/integration-tests/balance/test_saturated_balance.py \

          assert COMM_WORLD.size == 2

        $ pytest test_comms.py  # works, calls MPI under the hood

      \end{minted}

\begin{frame}[fragile]
	\frametitle{Supporting \main{} and/or \release{}}

	\pause

	\begin{description}
		\item[Strategy 1] Just pick one.
		\pause
		\item[Strategy 2] Maintain your own \release{} and \main{} branches.
		\pause
		\item[Strategy 3] Check the version.
			\begin{minted}[autogobble]{python}
if importlib.metadata.version("firedrake") > "2025.4":
    # new API
else:
    # old API
			\end{minted}
	\end{description}

	\pause

	We have Docker images for \release{} (\texttt{firedrake-vanilla-default:dev-release}) and \main{} (\texttt{firedrake-vanilla-default:dev-main}) that you can use for your CI.
\end{frame}
